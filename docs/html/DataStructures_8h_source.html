41/**
42 * @brief Represents a web (a live range variable) in the register allocation problem.
43 *
44 * Each web has a unique identifier, a set of live range intervals, a label,

46 */
47struct Web {

54 bool isDefAt(int line) const {

57 return false;

66 /**
67 * @brief Checks whether this web interferes with another.
68 * @details Two webs interfere if any of their live range intervals overlap,
69 * excluding boundary-only overlaps where one ends with '-' and the other
70 * starts with '+' at the same line (per spec non-interference rule).
71 *
72 * @par Time complexity
73 * O(I1 * I2 * P) where I1, I2 are the number of live range intervals in
74 * each web, and P is the number of program points per web (for marker lookups).
75 *
76 * @param other The other web to check against.
77 * @return True if the webs interfere, false otherwise.
78 */
79 bool interferesWith(const Web& other) const {
80 for (const auto& a : liveRanges) {
81 for (const auto& b : other.liveRanges) {
82 if (!a.overlaps(b)) continue;
83 if (a.end == b.start && isUseAt(a.end) && other.isDefAt(b.start)) continue;
84 if (a.start == b.end && isDefAt(a.start) && other.isUseAt(b.end)) continue;
85 return true;
86 }
87 }
88 return false;
89 }
90};
