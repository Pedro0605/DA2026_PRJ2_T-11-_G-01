49bool GraphColoring::basicColoring(Graph<int>& ig, int K,
50 std::unordered_map<int, int>& colorAssignment) {
51 std::unordered_set<int> active;
52 for (auto* v : ig.getVertexSet()) {
53 active.insert(v->getInfo());
54 }
55
56 std::stack<int> nodeStack;
57
58 while (!active.empty()) {
59 bool found = false;
60 for (auto it = active.begin(); it != active.end();) {
61 Vertex<int>* v = ig.findVertex(*it);
62 int deg = getEffectiveDegree(v, active);
63 if (deg < K) {
64 nodeStack.push(*it);
65 it = active.erase(it);
66 found = true;
67 } else {
68 ++it;
69 }
70 }
71 if (!found) {
72 return false;
73 }
74 }
75
76 while (!nodeStack.empty()) {
77 int id = nodeStack.top();
78 nodeStack.pop();
79
80 std::vector<bool> used(K, false);
81 Vertex<int>* v = ig.findVertex(id);
82 for (auto* e : v->getAdj()) {
83 int nid = e->getDest()->getInfo();
84 auto it = colorAssignment.find(nid);
85 if (it != colorAssignment.end() && it->second >= 0 && it->second < K) {
86 used[it->second] = true;
87 }
88 }
89
90 int color = -1;
91 for (int c = 0; c < K; c++) {
92 if (!used[c]) {
93 color = c;
94 break;
95 }
96 }
97 if (color == -1) {
98 return false;
99 }
100 colorAssignment[id] = color;
101 }
102
103 return true;
104}

197template <class T>
198std::map<T, std::string> GraphColoring::allocateRegistersFree(Graph<T>* graph, int maxRegisters) {
199 std::map<T, std::string> finalAllocation;
200 std::vector<Vertex<T>*> vertices = graph->getVertexSet();
201
202 if (vertices.empty()) return finalAllocation;
203
204 std::sort(vertices.begin(), vertices.end(), [](Vertex<T>* a, Vertex<T>* b) {
205 return a->getAdj().size() > b->getAdj().size();
206 });
207
208 std::map<T, int> colorMap;
209 for (auto* v : vertices) {
210 colorMap[v->getInfo()] = -1;
211 }
212
213 int currentColor = 0;
214 int uncoloredCount = vertices.size();
215
216 while (uncoloredCount > 0 && currentColor < maxRegisters) {
217 for (auto* v : vertices) {
218 if (colorMap[v->getInfo()] == -1) {
219 bool safeToColor = true;
220
221 for (auto* edge : v->getAdj()) {
222 if (colorMap[edge->getDest()->getInfo()] == currentColor) {
223 safeToColor = false;
224 break;
225 }
226 }
227
228 if (safeToColor) {
229 colorMap[v->getInfo()] = currentColor;
230 uncoloredCount--;
231 }
232 }
233 }
234 currentColor++;
235 }
236
237 for (auto* v : vertices) {
238 T webInfo = v->getInfo();
239 int assignedColor = colorMap[webInfo];
240
241 if (assignedColor == -1) {
242 finalAllocation[webInfo] = "M";
243 } else {
244 finalAllocation[webInfo] = "r" + std::to_string(assignedColor);
245 }
246 }
247
248 return finalAllocation;
249}

258static int pickWebToSplit(const Graph<int>& graph) {

272/**
273 * @brief Splits a web's live range at its midpoint.
274 * @details The original web keeps intervals and program points at or before the
275 * split point. A new web (with incremented ID and label suffixed with "_s")
276 * receives the remainder. Intervals spanning the split point are divided.
277 * @param webs The vector of all webs (the new web is appended).
278 * @param idx Index in @p webs of the web to split.
279 * @param nextId The next available web ID (incremented after creating the new web).
280 * @param entry Output tuple populated as (originalId, newWebId, splitPoint).
281 */
282static void splitWeb(std::vector<Web>& webs, size_t idx, int& nextId,
283 std::tuple<int,int,int>& entry) {
284 Web& original = webs[idx];
285
286 int splitPoint = -1;
287 if (original.liveRanges.size() > 1) {
288 for (size_t i = 0; i + 1 < original.liveRanges.size(); ++i) {
289 if (original.liveRanges[i].end + 1 < original.liveRanges[i + 1].start) {
290 splitPoint = original.liveRanges[i].end;
291 break;
292 }
293 }
294 }
295 if (splitPoint < 0) {
296 int minStart = original.liveRanges.front().start;
297 int maxEnd = original.liveRanges.front().end;
298 for (const auto& interval : original.liveRanges) {
299 if (interval.start < minStart) minStart = interval.start;
300 if (interval.end > maxEnd) maxEnd = interval.end;
301 }
302 splitPoint = (minStart + maxEnd) / 2;
303 }
304
305 Web newWeb;
306 newWeb.id = nextId++;
307 newWeb.label = original.label + "_s" + std::to_string(nextId);
308
309 std::vector<Interval> originalRanges, newRanges;
310 for (const auto& interval : original.liveRanges) {
311 if (interval.end <= splitPoint) {
312 originalRanges.push_back(interval);
313 } else if (interval.start > splitPoint) {
314 newRanges.push_back(interval);
315 } else {
316 originalRanges.push_back({interval.start, splitPoint});
317 newRanges.push_back({splitPoint + 1, interval.end});
318 }
319 }
320
321 std::vector<ProgramPoint> originalPoints, newPoints;
322 for (const auto& point : original.points) {
323 if (point.line <= splitPoint) {
324 originalPoints.push_back(point);
325 } else {
326 newPoints.push_back(point);
327 }
328 }
329
330 original.liveRanges = originalRanges;
331 original.points = originalPoints;
332 newWeb.liveRanges = newRanges;
333 newWeb.points = newPoints;
334
335 entry = std::make_tuple(original.id, newWeb.id, splitPoint);
336 webs.push_back(newWeb);
337}
