180/*
181 * Auxiliary function to add an outgoing edge to a vertex (this),
182 * with a given destination vertex (d) and edge weight (w).
183 */
184template <class T>
185Edge<T> * Vertex<T>::addEdge(Vertex<T> *d, double w) {
186 auto newEdge = new Edge<T>(this, d, w);
187 adj.push_back(newEdge);
188 d->incoming.push_back(newEdge);
189 return newEdge;
190}
191
192/*
193 * Auxiliary function to remove an outgoing edge (with a given destination (d))
194 * from a vertex (this).
195 * Returns true if successful, and false if such edge does not exist.
196 */
197template <class T>
198bool Vertex<T>::removeEdge(T in) {
199 bool removedEdge = false;
200 auto it = adj.begin();
201 while (it != adj.end()) {
202 Edge<T> *edge = *it;
203 Vertex<T> *dest = edge->getDest();
204 if (dest->getInfo() == in) {
205 it = adj.erase(it);
206 deleteEdge(edge);
207 removedEdge = true; // allows for multiple edges to connect the same pair of vertices (multigraph)
208 }
209 else {
210 it++;
211 }
212 }
213 return removedEdge;
214}

216/*
217 * Auxiliary function to remove an outgoing edge of a vertex.
218 */
219template <class T>
220void Vertex<T>::removeOutgoingEdges() {
221 auto it = adj.begin();
222 while (it != adj.end()) {
223 Edge<T> *edge = *it;
224 it = adj.erase(it);
225 deleteEdge(edge);
226 }
227}

264template <class T>
265void Vertex<T>::deleteEdge(Edge<T> *edge) {
266 Vertex<T> *dest = edge->getDest();
267 auto it = dest->incoming.begin();
268 while (it != dest->incoming.end()) {
269 if ((*it)->getOrig() == this) {
270 it = dest->incoming.erase(it);
271 }
272 else {
273 it++;
274 }
275 }
276 delete edge;
277}

341template <class T>
342bool Graph<T>::addVertex(const T &in) {
343 if (findVertex(in) != nullptr)
344 return false;
345 vertexSet.push_back(new Vertex<T>(in));
346 return true;
347}

349/*
350 * Removes a vertex with a given content (in) from a graph (this), and
351 * all outgoing and incoming edges.
352 * Returns true if successful, and false if such vertex does not exist.
353 */
354template <class T>
355bool Graph<T>::removeVertex(const T &in) {
356 for (auto it = vertexSet.begin(); it != vertexSet.end(); it++) {
357 if ((*it)->getInfo() == in) {
358 auto v = *it;
359 v->removeOutgoingEdges();
360 for (auto u : vertexSet) {
361 u->removeEdge(v->getInfo());
362 }
363 vertexSet.erase(it);
364 delete v;
365 return true;
366 }
367 }
368 return false;
369}

371template <class T>
372bool Graph<T>::addFlowEdge(const T &sourc, const T &dest, double w) {
373 auto v1 = findVertex(sourc);
374 auto v2 = findVertex(dest);
375 if (v1 == nullptr || v2 == nullptr)
376 return false;
377 auto e1 = v1->addEdge(v2, w);
378 auto e2 = v2->addEdge(v1, 0);
379 e1->setReverse(e2);
380 e2->setReverse(e1);
381 return true;
382}

387 if (v != nullptr)
388 v->removeOutgoingEdges();
389 }
390 for (auto* v : vertexSet) {
391 if (v != nullptr) {
392 delete v;
393 }
394 }
