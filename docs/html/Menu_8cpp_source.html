13void Menu::run() {
14 int option = -1;
15 while (option != 9) {
16 displayMenu();
17 std::string input;
18 std::getline(std::cin, input);
19 if (input.empty() && rangesLoaded && registersLoaded) {
20 runConfiguredAlgorithm();
21 continue;
22 }
23 std::stringstream ss(input);
24 ss >> option;
25 if (ss.fail()) {
26 option = -1;
27 }
28 switch (option) {
29 case 1: loadRanges(); break;
30 case 2: loadRegisters(); break;
31 case 3: runBasic(); break;
32 case 4: runSpilling(); break;
33 case 5: runSplitting(); break;
34 case 6: runCustom(); break;
35 case 7: writeOutput(); break;
36 case 8: displayResults(); break;
37 case 9: std::cout << "Exiting...\n"; break;
38 default: std::cout << "Invalid option. Please try again.\n"; break;
39 }
40 if (option != 9 && isatty(STDIN_FILENO)) {
41 std::cout << "\nPress Enter to continue...";
42 std::cin.get();
43 }
44 }
45}
